cmake_minimum_required(VERSION 3.8)
project(yam_ros2_control)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(ament_cmake REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(kdl_parser REQUIRED)
find_package(orocos_kdl REQUIRED)

add_library(yam_can_driver SHARED src/yam_can_driver.cpp)
add_library(yam_hardware_interface SHARED src/yam_hardware_interface.cpp)
add_library(yam_physical_hardware SHARED src/yam_physical_hardware.cpp)
add_library(yam_live_hardware SHARED src/yam_live_hardware.cpp)

foreach(t yam_can_driver yam_hardware_interface yam_physical_hardware yam_live_hardware)
  target_compile_features(${t} PUBLIC cxx_std_17)
  target_include_directories(${t} PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>)
endforeach()
foreach(t yam_hardware_interface yam_physical_hardware yam_live_hardware)
  ament_target_dependencies(${t} hardware_interface pluginlib rclcpp rclcpp_lifecycle)
  target_link_libraries(${t} yam_can_driver)
endforeach()
ament_target_dependencies(yam_live_hardware kdl_parser orocos_kdl)
pluginlib_export_plugin_description_file(hardware_interface yam_ros2_control.xml)
install(TARGETS yam_can_driver yam_hardware_interface yam_physical_hardware yam_live_hardware
  EXPORT export_yam_hardware_interface LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_yam_hardware_interface HAS_LIBRARY_TARGET)
ament_export_dependencies(hardware_interface pluginlib rclcpp rclcpp_lifecycle kdl_parser orocos_kdl)
ament_package()
